```sql
-- MAIN (ewhdfqwfwofivojtsizn)
select
  count(*) filter (where account_kind is null)                                   as ยังไม่จัดชนิด,
  count(*) filter (where account_kind = 'person' and employee_id is null)         as person_ยังไม่ผูก
from profiles;
select count(*) as person_ที่จะเสียตัวตนถ้าถอด_fallback
from profiles p join employees e on e.id = p.employee_id
where p.account_kind = 'person'
  and ( (p.line_id is not null and e.line_id is null)
     or (nullif(trim(p.team),'')    is not null and nullif(trim(e.team),'')    is null)
     or (nullif(trim(p.section),'') is not null and nullif(trim(e.section),'') is null) );
```
