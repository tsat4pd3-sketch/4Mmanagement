## 1. หลักการที่ห้ามหลุด

> **มิเตอร์ = ข้อเท็จจริง · คน = เหตุผล · ESM = เจ้าของสูตร**
> (หลักเดียวกับ SCADA — ถ้าหลุดข้อนี้ จะได้ตัวเลขพลังงาน 2 ชุดที่เถียงกันเอง)

1. **เก็บ "ค่าที่อ่านได้" ไม่ใช่ "ค่าที่คำนวณแล้ว"** — เก็บเลขมิเตอร์สะสม (reading) แล้วให้ ESM คิด kWh เอง
   ถ้าเก็บ delta ที่คนคิดมาแล้ว จะแก้ย้อนหลังไม่ได้เมื่อพบว่าจดผิด
2. **สูตรพลังงานทุกตัวอยู่ `src/utils/energy.js` ที่เดียว** — ตามกฎเดียวกับ `oee.js`
   **ห้ามเขียนสูตร kWh/ชิ้น หรือปันส่วนพลังงานเองในหน้าใดๆ**
3. **เริ่มจาก "จดมือ" ให้ใช้งานได้จริงก่อน** — โรงงานยังไม่มี IoT meter ทุกจุด
   ถ้าออกแบบโดยสมมติว่าต้องมีมิเตอร์อัตโนมัติก่อน โมดูลนี้จะไม่ได้ใช้เลย
4. **ประหยัดพลังงาน ≠ ลด kWh อย่างเดียว** — ค่าไฟไทยเป็น **TOU + demand charge**
   ลด kWh แต่ peak โผล่ = ค่าไฟไม่ลด · ต้องมองทั้ง 3 แกน: **หน่วย · ช่วงเวลา · พีค**
5. **ห้ามเดาการปันส่วน** — คอมเพรสเซอร์ 1 ตัวจ่าย 5 ไลน์ ปันยังไงเป็น **การตัดสินใจของคน**
   ระบบต้องให้เลือกวิธี และ**บอกบนจอเสมอว่าเลขนี้ปันมาด้วยวิธีไหน**

## 3. สถาปัตยกรรม — 3 เฟส เดินทีละขั้น

```
เฟส 1 · จดมือ (ทำได้เลย ไม่ต้องซื้ออะไร)
   ช่างจดเลขมิเตอร์ตามรอบ (รายวัน/รายกะ/รายเดือน)
   → ESM คิด kWh, ค่าไฟ, kWh/ชิ้น, เทียบ baseline
   ★ reuse ช่องอ่านค่าของ PM checklist ที่มีอยู่แล้ว (จุดตรวจชนิด "ค่าตัวเลข")

เฟส 2 · sub-meter รายโซน/รายเครื่องใหญ่
   ติดมิเตอร์ดิจิทัล (Modbus/pulse) เฉพาะจุดที่กินไฟเยอะ
   → gateway ยิงเข้า ESM ตามรอบ 15 นาที (เหมือน SCADA §3)
   → เห็นรายโซน + จับลมรั่ว/โหลดผีได้

เฟส 3 · realtime + demand
   มิเตอร์หลักส่ง demand 15 นาที → เตือนก่อนชน peak
   ★ ใช้ ingest pipeline ตัวเดียวกับ SCADA ไม่สร้างใหม่
```

> **ห้ามข้ามไปเฟส 3 ก่อน** — ถ้ายังไม่มีวินัยจดมิเตอร์ ข้อมูลอัตโนมัติก็ไม่มีใครดู

## 4. Schema (DR project — ที่เดียวกับ machines/mtn_orders)

```sql
-- ── จุดวัด (มิเตอร์) ─────────────────────────────────────────────
-- 1 แถว = 1 มิเตอร์จริงในโรงงาน · โครงเป็นต้นไม้ (มิเตอร์หลัก → ย่อย)
create table energy_meters (
  id            uuid primary key default gen_random_uuid(),
  code          text not null,              -- 'MDB-01', 'AIR-COMP-01'
  name          text not null,
  utility       text not null,              -- 'electric' | 'air' | 'water' | 'gas'  ★ data-driven ผ่าน energy_utilities
  unit          text not null,              -- 'kWh' | 'm3' | 'kg'
  parent_id     uuid references energy_meters(id),   -- ★ มิเตอร์แม่ (ใช้ตรวจ "ผลรวมลูก ≠ แม่" = รั่ว/มิเตอร์เสีย)
  machine_id    uuid references machines(id),        -- ★ ผูกด้วย id ไม่ใช่ machine_no (ชื่อเปลี่ยนได้)
  area_id       uuid references pm_facility_areas(id),
  line_name     text,                        -- ไลน์ผลิตที่มิเตอร์นี้จ่ายให้ตรงๆ (ถ้ามี)
  ct_ratio      numeric default 1,           -- ★ ตัวคูณหม้อแปลงกระแส — เลขหน้ามิเตอร์ × ct_ratio = หน่วยจริง
  digits        int,                         -- จำนวนหลักหน้าปัด (ใช้จับ rollover)
  read_cycle    text default 'daily',        -- 'shift' | 'daily' | 'monthly'
  is_active     boolean not null default true,
  team          text,                        -- mtn_teams.key · null = ของกลาง (กฎ "ใครเป็นเจ้าของ คนนั้นแก้")
  sort_order    int default 0,
  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now(),
  updated_by_name text,                      -- ★ ต้องมี → ใส่ใน DR_AUDIT_TABLES ได้
  unique (code)
);

-- ── ค่าที่อ่านได้ (ดิบ) ────────────────────────────────────────────
-- ★ เก็บ "เลขหน้าปัด" ไม่ใช่ผลต่าง — ผลต่างคำนวณตอนอ่าน แก้ย้อนหลังได้
create table energy_readings (
  id            uuid primary key default gen_random_uuid(),
  meter_id      uuid not null references energy_meters(id) on delete cascade,
  read_at       timestamptz not null,        -- เวลาที่อ่าน (เวลาไทย → ใช้ work_date_bangkok() เมื่อต้องการวันงาน)
  work_date     date not null,               -- ★ วันงานตัด 08:00 (กฎ Date/Time ของโปรเจค)
  shift         text,                        -- 'day' | 'night' | null (รอบรายเดือน)
  reading       numeric not null,            -- เลขหน้าปัดดิบ (ยังไม่คูณ ct_ratio)
  delta_raw     numeric,                     -- คำนวณตอน insert: reading - ก่อนหน้า (จัดการ rollover แล้ว)
  source        text not null default 'manual',   -- ★ 'manual' | 'meter' | 'import'  ต้องมีตั้งแต่แถวแรก
  is_reset      boolean not null default false,   -- มิเตอร์ถูกรีเซ็ต/เปลี่ยนตัว → delta ของแถวนี้ไม่นับ
  note          text,
  read_by_name  text,
  created_at    timestamptz not null default now(),
  unique (meter_id, read_at)
);
create index on energy_readings (meter_id, work_date);

-- ── ค่าพลังงานรายวัน (rollup) ────────────────────────────────────
-- ★ ตารางนี้คือสิ่งที่หน้าจอ "อ่าน" — ห้ามให้จอไปไล่ readings ดิบเอง (ดู §5.2)
create table energy_daily (
  meter_id      uuid not null references energy_meters(id) on delete cascade,
  work_date     date not null,
  qty           numeric not null,            -- หน่วยที่ใช้จริง (คูณ ct_ratio แล้ว)
  qty_peak      numeric,                     -- แยกตามช่วง TOU (เฟส 2+)
  qty_offpeak   numeric,
  cost          numeric,                     -- บาท (คำนวณจาก energy_tariffs)
  is_estimated  boolean not null default false,  -- ★ วันที่ไม่มีคนจด = เกลี่ยจากช่วง ต้องบอกว่าเป็นค่าประมาณ
  primary key (meter_id, work_date)
);

-- ── อัตราค่าไฟ (TOU) ─────────────────────────────────────────────
-- ★ ไม่ hardcode ค่าไฟในโค้ด — ขึ้นราคาแล้วแก้ตารางนี้ ไม่ต้อง deploy
create table energy_tariffs (
  id            uuid primary key default gen_random_uuid(),
  utility       text not null default 'electric',
  name          text not null,               -- 'TOU 22-33kV', 'อัตราปกติ'
  effective_from date not null,              -- ★ มีผลตั้งแต่วันไหน (เก็บประวัติ ห้ามแก้ทับ)
  peak_rate     numeric,                     -- บาท/kWh ช่วง peak
  offpeak_rate  numeric,
  flat_rate     numeric,                     -- ใช้เมื่อไม่แยก TOU
  demand_rate   numeric,                     -- บาท/kW (ค่าความต้องการพลังไฟฟ้า)
  ft_rate       numeric,                     -- ค่า Ft (บาท/kWh) เปลี่ยนทุก 4 เดือน
  service_fee   numeric,                     -- ค่าบริการรายเดือน
  is_active     boolean not null default true
);

-- ── ช่วงเวลา TOU (peak/off-peak) — data-driven ห้าม hardcode ────
create table energy_tou_windows (
  id            uuid primary key default gen_random_uuid(),
  tariff_id     uuid not null references energy_tariffs(id) on delete cascade,
  day_kind      text not null,               -- 'weekday' | 'weekend_holiday'
  period        text not null,               -- 'peak' | 'offpeak'
  start_time    time not null,
  end_time      time not null
);

-- ── เป้าหมายประหยัด (ต่อมิเตอร์/ต่อโรงงาน) ────────────────────────
create table energy_targets (
  id            uuid primary key default gen_random_uuid(),
  meter_id      uuid references energy_meters(id),   -- null = ทั้งโรงงาน
  month_key     text not null,               -- 'YYYY-MM'
  target_qty    numeric,                     -- หน่วยเป้า
  target_sec    numeric,                     -- kWh ต่อชิ้นเป้า (Specific Energy Consumption)
  baseline_from text,                        -- 'YYYY-MM' ที่ใช้เป็นฐานเทียบ
  note          text,
  unique (meter_id, month_key)
);
```

## 5. ⚠️ ปัญหาที่ต้องแก้ก่อนเขียนโค้ด

### 5.1 มิเตอร์สะสมจะ rollover / reset — delta ติดลบ
หน้าปัด 6 หลักวนกลับที่ 999999 → 0 · เปลี่ยนมิเตอร์ใหม่เริ่มนับ 0 · ช่างกดรีเซ็ต
**ถ้าคิด `delta = ใหม่ − เก่า` ตรงๆ จะได้ค่าติดลบมหาศาล แล้วกราฟทั้งเดือนพัง**

**กติกา:**
- `delta < 0` → ถ้า `digits` ตั้งไว้ ให้ลอง rollover: `delta = (10^digits − เก่า) + ใหม่`
- ถ้าผลยังไม่สมเหตุผล (เกินค่าปกติ N เท่า) → **ไม่เดา** ตั้ง `is_reset = true`, `delta_raw = null`
  แล้ว**ขึ้นเตือนให้คนยืนยัน** ว่าเปลี่ยนมิเตอร์ หรือจดผิด
- หลักเดียวกับ SCADA §5.7 (counter reset) — **ห้ามกลืน error เงียบ**

### 5.2 ปริมาณข้อมูล (บทเรียนจาก SCADA §5.1)
เฟส 3 มิเตอร์ 30 ตัว × ทุก 15 นาที = **1.05 ล้านแถว/ปี** · DR ตอนนี้ใช้ไป 36MB จาก 500MB
**ต้องมี rollup ตั้งแต่วันแรก ไม่ใช่ค่อยมาทำตอน DB เต็ม:**
- `energy_readings` = ข้อมูลดิบ **เก็บ 90 วัน** แล้วลบ (มี `energy_daily` แล้ว)
- `energy_daily` = เก็บถาวร (30 มิเตอร์ × 365 = 11k แถว/ปี — เล็กมาก)
- **หน้าจอทุกหน้าอ่าน `energy_daily` เท่านั้น** ห้าม query readings ดิบ

### 5.3 kWh/ชิ้น (SEC) — ตัวหารพังง่ายมาก
- **งานคู่ LH/RH**: 1 stroke ได้ 2 ชิ้น → ต้องใช้ `pairAwareTotal` (กฎเดิมของโปรเจค)
  ไม่งั้น SEC ดูดีขึ้น 2 เท่าโดยไม่มีอะไรเปลี่ยน
- **วันไม่ผลิตแต่กินไฟ** → ตัวหาร = 0 → **ห้ามหาร ต้องแสดงเป็น "โหลดตอนไม่ผลิต (kWh)" แทน**
  (นี่คือตัวเลขที่มีค่าที่สุดของโมดูลนี้ — ดู §5.6)
- **หลายพาร์ทในกะเดียว** ใช้ไฟไม่เท่ากัน → SEC ระดับไลน์เท่านั้น **อย่าลงไปถึงรายพาร์ทจนกว่าจะมีมิเตอร์รายเครื่อง**

### 5.4 การปันส่วนพลังงานส่วนกลาง — ต้องให้คนเลือก ห้ามเดา
คอมเพรสเซอร์ 1 ตัวจ่าย 5 ไลน์ → จะบอกว่า "ไลน์ A ใช้ไฟเท่าไหร่" ต้องปัน
**มี 3 วิธี ผลต่างกันมาก:**

| วิธี | เหมาะกับ | ข้อควรระวัง |
|---|---|---|
| เฉลี่ยเท่ากันทุกไลน์ | เริ่มต้น ไม่มีข้อมูล | ไลน์ที่ใช้ลมน้อยโดนคิดเกิน |
| **ตามชั่วโมงเดินเครื่อง** (จาก `production_sessions`) | ค่าเริ่มต้นที่แนะนำ | ไลน์เดินแต่ไม่ใช้ลมก็โดนคิด |
| ตามยอดผลิต (`pairAwareTotal`) | ไลน์ที่พลังงานแปรผันตามชิ้น | วันผลิตน้อยดูเหมือนประหยัด ทั้งที่โหลดผีเท่าเดิม |

→ เก็บวิธีที่เลือกไว้ที่ `facility_supply_links` (เพิ่ม `alloc_method` + `alloc_weight`)
→ **ทุกจอที่โชว์ตัวเลขปันส่วน ต้องมีป้าย "ปันตามชั่วโมงเดินเครื่อง" กำกับ** ห้ามโชว์เลขลอยๆ เหมือนเป็นค่าที่วัดได้จริง

### 5.5 ค่าไฟ ≠ kWh × เรตเดียว
ค่าไฟโรงงานไทย = `(peak×เรต_peak + offpeak×เรต_offpeak) + demand_kW×เรต_demand + Ft×หน่วยรวม + ค่าบริการ + VAT`
- **demand charge คิดจาก kW สูงสุด 15 นาทีของเดือน** — เดินเครื่องใหญ่พร้อมกันครั้งเดียวทั้งเดือน ค่าไฟขึ้นทั้งเดือน
- **ลด kWh แต่ peak ยังชน = บิลไม่ลด** → จอต้องแยก 2 ตัวเลขนี้ให้ชัด
- **Ft เปลี่ยนทุก 4 เดือน** → `effective_from` เก็บประวัติ **ห้ามแก้ทับแถวเดิม** (ไม่งั้นค่าไฟย้อนหลังเปลี่ยนตาม = ขัดหลักเดียวกับ "ห้าม recompute OEE กะเก่าด้วย master ปัจจุบัน")

### 5.6 "โหลดผี" (baseload) คือของจริงที่ควรจับก่อนอย่างอื่น
พลังงานตอน**ไม่มีการผลิต** = ลมรั่ว · ไฟส่องสว่างลืมปิด · เครื่องเดิน idle · คูลลิ่งเดินเปล่า
**ตรวจได้จากข้อมูลที่มีอยู่แล้ว ไม่ต้องมีมิเตอร์รายเครื่อง:**
```
baseload = kWh ของช่วงที่ไม่มี production_sessions เปิด (วันหยุดจาก company_calendar / ช่วงกะปิด)
```
- โรงงานทั่วไป baseload = 15-30% ของทั้งหมด · **ถ้าเกินนั้นคือมีของรั่ว**
- **นี่คือ quick win ของเฟส 1** — จดมิเตอร์วันหยุด 1 วันก็เห็นแล้ว
- เจอแล้วเปิด **ใบซ่อม MO** ได้เลย (ลักษณะปัญหา "ลมรั่ว" มีอยู่แล้วใน master)

### 5.7 คนลืมจด / จดผิด
- **ลืมจด** → วันนั้นไม่มีแถว → `energy_daily.is_estimated = true` เกลี่ยจากช่วงที่มี **และต้องแสดงบนจอว่าเป็นค่าประมาณ** (หลักเดียวกับ "~ จะส่งต่อกะหน้า" ใน Daily Report)
- **จดผิดหลัก** (พิมพ์ 12345 เป็น 1234) → delta กระโดด → เตือนเมื่อเกินค่าปกติ N เท่า ให้ยืนยันก่อนบันทึก
- **จดย้อนหลัง** → อนุญาต แต่ต้อง recompute `energy_daily` ของช่วงนั้นใหม่

### 5.8 นาฬิกา 2 แหล่ง (บทเรียนเดิมของโปรเจค)
มิเตอร์ดิจิทัลมีนาฬิกาของตัวเอง — เก็บ `read_at` (จากมิเตอร์) **คู่กับ** `created_at` (server) เสมอ
เคยกัดมาแล้วที่ `prod_orders` (ปิดก่อนเปิด) และมี trigger guard อยู่

## 12. 🌱 Carbon — จาก kWh ไป CO2e และเรื่อง "carbon credit"

> คำถาม user (2026-08-18): *"เรื่อง feature การใช้พลังงาน เราต่อยอดไปถึง carbon credit ด้วยได้มั้ย"*
> **คำตอบสั้น: ได้ — แต่ต้องแยกให้ขาดก่อนว่ากำลังพูดถึงอันไหน เพราะ 3 อันนี้คนละเรื่องกันคนละระดับความยาก**

### 12.1 ⚠️ 3 คำที่มักถูกใช้ปนกัน (ต้องแยกก่อนคุยต่อ)

| | คืออะไร | ต้องมีอะไร | ระบบเราทำได้แค่ไหน |
|---|---|---|---|
| **① Carbon Footprint องค์กร (CFO)** | *เราปล่อยเท่าไหร่* | ข้อมูลกิจกรรม (kWh/ลิตร) × emission factor | ✅ **ทำได้เลยด้วยข้อมูลที่มีอยู่** (Scope 2 จาก `energy_monthly`) |
| **② Carbon Footprint ผลิตภัณฑ์ (CFP)** | *1 ชิ้นปล่อยกี่ kgCO2e* | ① + ยอดผลิตราย MAT + การปันส่วน | ✅ ทำได้บนพาร์ทต้นแบบ (มี `prod_orders` ราย mat ครบแล้ว) |
| **③ Carbon Credit (T-VER)** | *ใบรับรองว่า "ลดได้" ที่เอาไปขาย/ชดเชยได้* | โครงการจดทะเบียนกับ **TGO** + methodology ที่อนุมัติ + monitoring plan + **ผู้ทวนสอบภายนอก (VVB)** + TGO รับรอง | ❌ **ระบบออกเครดิตให้ไม่ได้** — ทำได้แค่เป็น *"ระบบเก็บหลักฐาน/monitoring"* ให้โครงการที่คนไปจดทะเบียนเอง |

**หัวใจที่มักเข้าใจผิด: "ประหยัดไฟได้ 10% = ได้คาร์บอนเครดิต" — ไม่จริง**
เครดิตไม่ได้เกิดจากการที่ตัวเลขลดลง แต่เกิดจาก **โครงการที่ถูกจดทะเบียนและทวนสอบแล้ว** ซึ่งต้องผ่าน:
1. **Additionality** — ต้องพิสูจน์ว่าถ้าไม่มีโครงการนี้ จะไม่เกิดการลดนี้ (การเปลี่ยนหลอดเป็น LED ที่ยังไงก็คุ้มอยู่แล้ว มักตกข้อนี้)
2. **Baseline ตาม methodology ที่อนุมัติ** — ไม่ใช่ "เทียบเดือนก่อน" แบบที่หน้าจอเราทำ
3. **Monitoring plan** — ต้องระบุไว้ล่วงหน้าว่าจะวัดด้วยอะไร ความถี่เท่าไหร่ สอบเทียบเมื่อไหร่
4. **ทวนสอบโดยบุคคลที่สาม (VVB)** ทุกรอบ แล้ว TGO ถึงออกเครดิต

### 12.2 💸 เครดิตคุ้มไหม — ทำเลขให้ดูก่อนตัดสินใจ

**สูตร:** `tCO2e ที่ลดได้ = kWh ที่ลดได้ × EF(kgCO2e/kWh) ÷ 1000`
ด้วย EF ไฟฟ้าไทยระดับ **~0.5 kgCO2e/kWh** (ตัวเลขจริงต้องเอาจากประกาศ TGO ปีนั้น — **ห้าม hardcode** ดู §12.4)

| ถ้าลดได้ | = tCO2e/ปี | มูลค่าเครดิตโดยประมาณ |
|---|---|---|
| 500,000 kWh/ปี | ~250 | หลักหมื่นบาท/ปี |
| 6,000,000 kWh/ปี | ~3,000 | หลักแสนบาท/ปี |

เทียบกับ **ค่าทวนสอบ VVB ระดับหลักแสนบาทต่อรอบ** (ต้องขอใบเสนอราคาจริง) + ค่าที่ปรึกษาทำเอกสาร
→ **โรงงานเดียวทำโครงการประหยัดพลังงานเพื่อขายเครดิต มักไม่คุ้ม** (แม้แต่โซลาร์รูฟ 1 MW ก็ให้ราว 700 tCO2e/ปีเท่านั้น)
> ⚠️ ราคาเครดิต T-VER ในไทยที่ผ่านมาอยู่ระดับ **หลักสิบถึงหลักร้อยบาท/tCO2e** และผันผวนมาก — **ต้องเช็คราคาปัจจุบันก่อนใช้ตัดสินใจ ห้ามอ้างตัวเลขในเอกสารนี้เป็นราคาจริง**

**เงื่อนไขที่ทำให้กลับมาคุ้ม (ควรกลับมาดูเมื่อเข้าเงื่อนไขข้อใดข้อหนึ่ง):**
- ติด **โซลาร์รูฟขนาดใหญ่** (ปริมาณต่อโครงการเยอะกว่างานประหยัดไฟมาก)
- **รวมหลายโรงงานในกลุ่ม TSG เป็นโครงการเดียว** — ค่าทวนสอบหารกันได้ · ระบบนี้ออกแบบมาเพื่อ rollout ทั้งกลุ่มอยู่แล้ว (ดู `/group-overview`) จึงเป็นทางที่สมเหตุผลที่สุด
- ลูกค้า/บริษัทแม่ **ต้องการเครดิตเพื่อชดเชยของตัวเอง** (มูลค่าอยู่ที่ความสัมพันธ์ ไม่ใช่ราคาตลาด)

### 12.3 🎯 เป้าที่คุ้มกว่าและควรทำก่อน: **ตอบลูกค้าให้ได้ ไม่ใช่ขายเครดิต**

- **Scope 1+2 ของเรา = Scope 3 ของลูกค้า** — OEM (Ford ฯลฯ) ทยอยขอข้อมูลการปล่อยจาก supplier
  ของที่เขาขอคือ **CFO รายปี** และ **CFP ต่อพาร์ท** ไม่ใช่เครดิต
- **CFP ต่อพาร์ทเป็นของที่เราได้เปรียบ** — ระบบมียอดผลิตราย MAT ราย
  ไลน์ครบอยู่แล้ว (`prod_orders` + `pairAwareTotal` + ชั้น OP) จึงหาร kWh ลงพาร์ทได้จริง
  ไม่ต้องเดาแบบที่โรงงานส่วนใหญ่ต้องทำ
- **ต่อยอด ISO 14001 / ISO 50001 / เกณฑ์ พพ.** ที่โรงงานมักต้องรายงานอยู่แล้ว
- **ลดค่าไฟจริงคือเงินที่ได้แน่นอนกว่าเครดิตมาก** — โมดูลพลังงานเฟส 2 ให้ผลตอบแทนตรงนี้อยู่แล้ว

> **ข้อสรุปเชิงลำดับงาน: ทำ ① CFO → ② CFP ก่อน · ③ เครดิต เก็บไว้เป็นทางเลือกตอนกลุ่ม TSG พร้อมรวมโครงการ**
> การทำ ①② ไม่เสียเปล่าแม้ไม่ทำ ③ เลย เพราะเป็นฐานข้อมูลชุดเดียวกันที่โครงการเครดิตต้องใช้อยู่ดี

> ### ✅ ขอบเขตที่ user ตัดสินแล้ว (2026-08-19)
> *"หลักๆ ถ้าลูกค้าถามและเราตอบได้ scope 1+2 ไม่ต้องประเมินถึง end user"*
> **→ เป้าของโมดูลนี้คือ "ตอบลูกค้าให้ได้" เท่านั้น · ทำ Scope 1 + 2 · ไม่ประเมิน Scope 3**
> เป็นการตัดสินใจที่ถูกและตรงกับมาตรฐาน — GHG Protocol กำหนดให้ **Scope 1+2 เป็นส่วนที่บังคับรายงาน
> ส่วน Scope 3 เป็นทางเลือก** และของที่ลูกค้าเอาไปใช้จริงคือ Scope 1+2 ของเรา (= Scope 3 ของเขา)
>
> **⚠️ ข้อเดียวที่ต้องรู้ล่วงหน้า กันเจอหน้างานแล้วตอบไม่ได้:**
> ถ้าลูกค้าถาม **"CFO ของบริษัท"** → Scope 1+2 ตอบได้ **ครบ 100%** ✅
> ถ้าลูกค้าถาม **"คาร์บอนต่อชิ้น (CFP)"** → Scope 1+2 ให้ได้แค่ **ช่วง gate-to-gate (เฉพาะที่เกิดในรั้วโรงงานเรา)**
> ส่วนที่ขาดคือ **คาร์บอนของเหล็ก/พาร์ทที่เราซื้อเข้ามา** ซึ่งมักเป็นก้อนใหญ่ที่สุดของชิ้นงานปั๊มโลหะ
> **ทางที่ถูกคือ "ขอตัวเลขจาก supplier" ไม่ใช่ "เราประเมินให้"** — กฎ §12.5 ข้อ 5 ยังยืน: **ห้ามเดา Scope 3**
> → หน้าจอ/ใบรายงาน CFP **ต้องเขียนกำกับเสมอว่าครอบคลุมแค่ช่วงไหน** ไม่งั้นลูกค้าเอาไปใช้ผิด แล้วย้อนกลับมาที่เรา

### 12.4 Schema (DR project — additive ทั้งหมด)

```sql
-- ── ชนิดพลังงาน/เชื้อเพลิง (data-driven — ห้าม hardcode รายการใน UI) ──
create table energy_utilities (
  key         text primary key,          -- 'electric' | 'diesel' | 'lpg' | 'gasoline'
  label       text not null,
  unit        text not null,             -- 'kWh' | 'litre' | 'kg'
  ghg_scope   int,                       -- 1 = เผาไหม้เอง · 2 = ซื้อไฟฟ้า · null = ยังไม่จัด
  icon        text, sort_order int default 0, is_active boolean not null default true
);

-- ── ค่าการปล่อย (Emission Factor) ──
-- ★ 1 แถว = EF ของ 1 ชนิด ณ ช่วงเวลาหนึ่ง — เก็บประวัติ ห้ามแก้ทับ
create table energy_emission_factors (
  id             uuid primary key default gen_random_uuid(),
  utility_key    text not null references energy_utilities(key),
  kg_co2e        numeric not null,       -- kgCO2e ต่อ 1 หน่วยของ utility นั้น
  effective_from date not null,          -- ★ EF เปลี่ยนเกือบทุกปี
  source_ref     text not null,          -- ★★ 'TGO Emission Factor ฉบับ เม.ย. 2567' — ไม่มีที่มา = ทวนสอบไม่ผ่าน
  note           text,
  updated_by_name text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (utility_key, effective_from)
);
```

**ไม่ต้องมีตาราง "carbon" แยก** — CO2e เป็น **ค่าที่คำนวณจาก `energy_monthly` ที่มีอยู่แล้ว**
(ปริมาณ × EF ของเดือนนั้น) เก็บซ้ำลง DB = มีตัวเลข 2 ชุดที่ drift กันได้ ผิดหลัก single source of truth
**Scope 1 (น้ำมันรถยก / เครื่องปั่นไฟ / แก๊ส) ใช้ `energy_monthly` แถวเดิมได้เลย** แค่เปลี่ยน `utility`
— คอลัมน์นี้ถูกใส่ไว้ใน unique key ตั้งแต่เฟส 1 เผื่อกรณีนี้พอดี (ดูคอมเมนต์ใน migration)

**สูตรทั้งหมดไปอยู่ `src/utils/energy.js` ไฟล์เดิม ห้ามแตกไฟล์ `carbon.js` ใหม่**
(`efFor(utility, date, factors)` · `co2eOf(qty, ef)` · `co2ePerPiece(...)`)
`energy_utilities` / `energy_emission_factors` ต้องเข้า **`DR_AUDIT_TABLES`** (master ที่แก้แล้วเลขทั้งระบบขยับ)

### 12.5 ⚠️ กฎเหล็กของงานคาร์บอน (ผิดข้อไหนก็ทำให้ตัวเลขใช้อ้างอิงไม่ได้)

1. **EF ต้องผูกกับ "ช่วงเวลา" และต้องอ้างที่มาได้เสมอ** (`effective_from` + `source_ref`)
   **ห้ามแก้ EF ทับของเก่า และห้าม recompute CO2e ของปีที่รายงานไปแล้วด้วย EF ปีปัจจุบัน**
   — หลักเดียวกับ *"ห้าม blanket-recompute OEE กะเก่าด้วย master ปัจจุบัน"* · ตัวเลขที่รายงานลูกค้า/หน่วยงานไปแล้วต้องนิ่ง
2. **🔴 ห้ามใช้คำว่า "คาร์บอนเครดิต" บนหน้าจอกับตัวเลขที่ยังไม่ผ่านการทวนสอบ**
   ใช้คำว่า **"ปริมาณที่ลดได้ (คำนวณเอง · ยังไม่ผ่านการทวนสอบ)"** เท่านั้น
   ผู้บริหารเห็นคำว่า "เครดิต" แล้วเข้าใจว่าเป็นทรัพย์สินที่ขายได้ → เอาไปพูดต่อข้างนอก = **กลายเป็นการกล่าวอ้างเท็จ (greenwashing)**
   ป้ายต้องแรงเท่าป้าย 🧪 จำลอง ใน `/group-overview` และป้าย ✍️ กรอกมือ ในโมดูลพลังงาน
3. **ไม่มี EF ของช่วงเวลานั้น → คืน `null` ห้ามหยิบ EF ปีอื่นมาใช้เงียบๆ**
   (กฎเดิมของโปรเจค: *ประเมินไม่ได้ = null + บอกเหตุผล ห้ามแปลงเป็น 0*)
4. **ต้องพก "ชั้นคุณภาพข้อมูล" ไปด้วยเสมอ** — `energy_monthly.source` (`manual`/`meter`/`estimated`) มีอยู่แล้ว
   รายงานต้องบอกว่ากี่ % ของตัวเลขมาจากมิเตอร์จริง · **ตัวเลขที่กรอกมือล้วนใช้ยื่นทวนสอบไม่ได้**
5. **Scope 1 / 2 / 3 ห้ามบวกรวมกันโดยไม่บอกว่ารวมอะไรบ้าง** — และ **ห้ามรายงาน Scope 3 จนกว่าจะมีข้อมูลจริง**
   (เดา Scope 3 = ตัวเลขที่ผิดมากที่สุดในรายงานคาร์บอนทั่วโลก)
6. **ห้ามอ้าง "carbon neutral / net zero" จากตัวเลขในระบบนี้** — ต้องมีการชดเชยที่ทวนสอบแล้วเท่านั้น

### 12.6 ระบบเรามีของดีอะไรอยู่แล้วสำหรับงานนี้ (จุดแข็งจริง ไม่ใช่ขายฝัน)

| มีอยู่แล้ว | ใช้กับงานคาร์บอนยังไง |
|---|---|
| **`audit_log` + `updated_by_name`** | ผู้ทวนสอบถามข้อแรกเสมอว่า *"ใครแก้ตัวเลขนี้ เมื่อไหร่"* — เรามีคำตอบระดับรายฟิลด์อยู่แล้ว |
| **`energy_monthly.source`** | ชั้นคุณภาพข้อมูล (manual/meter/estimated) ที่ CFO ต้องรายงาน — ออกแบบไว้ตั้งแต่เฟส 1 |
| **`improvements` (Kaizen)** | โครงสร้าง *baseline ก่อน → หลัง + วันเริ่ม + หน้าต่างเทียบ* คือรูปเดียวกับที่โครงการเครดิตต้องใช้ · เพิ่ม `problem_source='energy'` แล้วได้ทั้งบาทและ tCO2e จากโปรเจคเดียวกัน |
| **`prod_orders` ราย MAT + `pairAwareTotal` + ชั้น OP** | หาร kWh ลงเป็น **kgCO2e ต่อชิ้นรายพาร์ท** ได้จริง = ของที่ OEM ขอ |
| **`company_calendar`** | แยกวันผลิต/วันหยุด → baseload ซึ่งเป็นตัวชี้เป้าการลดที่ทำได้จริงที่สุด |
| **`transport_vehicles` / `kanban_delivery_rounds`** | ฐานสำหรับ Scope 1 (น้ำมันรถยก) และ Scope 3 ขนส่ง — ยังไม่มีข้อมูลน้ำมัน ต้องเริ่มบันทึกก่อน |
| **`doc_forms`** | ใบรายงาน CFO/CFP ต้อง register ตามกฎเอกสาร (แก้เลขฟอร์ม/ผู้เซ็นได้เองโดยไม่แก้โค้ด) |

### 12.7 เฟสของงานคาร์บอน

| เฟส | ทำอะไร | เงื่อนไขก่อนเริ่ม |
|---|---|---|
| **C1 — ตีเป็น CO2e** | `energy_utilities` + `energy_emission_factors` · การ์ด **tCO2e** ในหน้า `/energy` และบนผังรวม · ใบรายงานรายเดือน | ทำได้ทันทีบนข้อมูลรายเดือนที่มี · **ต้องมีคนหาประกาศ EF ของ TGO มากรอก** |
| **C2 — ครบ Scope 1 + CFP gate-to-gate** | เพิ่มน้ำมัน/แก๊สเป็น `utility` · kgCO2e ต่อชิ้นบนสายต้นแบบ 060/061 (**ระบุขอบเขตบนใบเสมอ**) | ต้องเริ่มบันทึกน้ำมันรถยก/เครื่องปั่นไฟ + ตกลงวิธีปันส่วน (**ห้ามเดา** ตาม §5.4) |
| **C3 — พร้อมให้ทวนสอบ** | log การสอบเทียบมิเตอร์ · export ชุดหลักฐาน · แยกชั้นคุณภาพข้อมูลในรายงาน | **ต้องมีมิเตอร์จริง (เฟส 2 ของพลังงาน) ก่อน** — กรอกมือล้วนยื่นไม่ผ่าน |
| **C4 — โครงการเครดิต** | ระบบเป็นแค่ *monitoring system* ของโครงการที่คนไปจดทะเบียนเอง | ต้องมีโครงการจริง + ตัดสินใจระดับกลุ่ม TSG (§12.2) |

> **C1 คือเฟสเดียวที่ทำได้เลยตอนนี้** — ที่เหลือติดเงื่อนไขข้อมูลจริง (น้ำมัน / มิเตอร์ / โครงการ)
> ทั้ง C1-C2 ให้คำตอบที่ลูกค้าถามครบตามขอบเขตที่ตัดสินไว้แล้ว (§12.3) โดยไม่ต้องแตะ Scope 3 เลย

### 12.8 สิ่งที่ตั้งใจ "ไม่ทำ"

- ❌ **ไม่ประเมิน Scope 3 / ไม่ไล่ถึง end user** — **คำสั่ง user 2026-08-19** (*"ตอบได้ scope 1+2 ไม่ต้องประเมินถึง end user"*)
  ตรงกับ GHG Protocol ที่บังคับแค่ Scope 1+2 · **ห้ามใครมาเติม Scope 3 แบบเดาๆ ทีหลัง** ถ้าลูกค้าขอคาร์บอนของวัสดุ = **ไปขอจาก supplier**
- ❌ **ไม่ออกเครดิต / ไม่ทำระบบซื้อขายเครดิตเอง** — เป็นอำนาจของ TGO และตลาด ไม่ใช่ของ ESM
- ❌ **ไม่เดา Scope 3** จากค่าเฉลี่ยอุตสาหกรรม — รายงานเลขที่เดามาแย่กว่าไม่รายงาน
- ❌ **ไม่ hardcode EF ในโค้ด** แม้แต่ค่าเดียว (เปลี่ยนทุกปี + ต้องอ้างที่มาได้)
- ❌ **ไม่คำนวณ CO2e จากเดือนที่ยังกรอกข้อมูลไม่ครบ** แล้วปล่อยให้ดูเหมือนยอดทั้งเดือน — ต้องบอกว่ากรอกไปกี่ % ของพื้นที่

### 12.9 คำถามที่ต้องได้คำตอบก่อนเริ่ม C1

**ต้องมีก่อนถึงจะได้ตัวเลข (ระบบสร้างเองไม่ได้):**
1. **ประกาศ EF ของ TGO** — ค่า kgCO2e/kWh ของไฟฟ้าจากระบบสายส่ง + **ปีที่ประกาศ** (ต้องกรอกทั้งค่าและที่มา §12.5 ข้อ 1)

**ตอบทีหลังได้ แต่ตอบแล้วงานง่ายขึ้น:**
2. **ใครเป็นเจ้าของเรื่องคาร์บอนในบริษัท** — ซ่อมบำรุง / ISO-QMR / ความปลอดภัย-สิ่งแวดล้อม? (ตัดสินสิทธิ์ `carbon:*`)
3. **เคยทำ CFO / ขึ้นทะเบียนกับ TGO แล้วหรือยัง** — ถ้าเคย ต้องใช้ EF และขอบเขตชุดเดิมเพื่อให้เทียบปีต่อปีได้
4. **ลูกค้าเคยขอข้อมูลการปล่อยมาหรือยัง** (Ford/OEM) — ถ้าเคย ขอแบบฟอร์มที่เขาส่งมา จะได้ทำให้ตรงช่องที่เขาถาม
   (สำคัญเป็นพิเศษหลังตัดสินขอบเขต §12.3 — ต้องรู้ว่าเขาถาม CFO หรือ CFP เพราะ 2 อันนี้เราตอบได้ไม่เท่ากัน)
5. **มีเชื้อเพลิงอะไรใช้ในโรงงานบ้าง** — รถยก (ดีเซล/LPG/ไฟฟ้า) · เครื่องปั่นไฟสำรอง · หม้อไอน้ำ · แก๊สเชื่อม (→ C2)
6. **มีโซลาร์รูฟ หรือแผนจะติดไหม** — เปลี่ยนคำตอบเรื่องเครดิตทั้งหมด (§12.2)

> **ข้อ 1 ไม่ใช่ตัวบล็อกการเขียนโค้ด** — สร้างตาราง EF เปล่าไว้ได้ แล้วหน้าจอขึ้น
> *"ยังไม่ได้ตั้งค่าการปล่อย — กรอกที่ ⚙️ ตั้งค่า"* ตามกฎ *ไม่มี EF = null + บอกเหตุผล ห้ามเดา*
> วันที่มีคนเอาตัวเลข TGO มากรอก ตัวเลข tCO2e ย้อนหลังทุกเดือนจะขึ้นเองทันที (คำนวณสด ไม่ได้เก็บซ้ำ)
